<?php

require_once "db.php";

$request_number = $_GET["id"] ?? $_POST["request_number"] ?? "";

if ($request_number === "") {
    die("رقم الطلب غير موجود.");
}


/* =========================
   حفظ التعديلات
========================= */

if ($_SERVER["REQUEST_METHOD"] === "POST") {

    $full_name = trim($_POST["full_name"] ?? "");
    $employee_number = trim($_POST["employee_number"] ?? "");
    $category = trim($_POST["category"] ?? "");
    $rank = trim($_POST["rank"] ?? "");
    $unit = trim($_POST["unit"] ?? "");
    $location = trim($_POST["location"] ?? "");
    $notes = trim($_POST["notes"] ?? "");

    if (
        $full_name === "" ||
        $employee_number === "" ||
        $category === "" ||
        $rank === "" ||
        $unit === "" ||
        $location === ""
    ) {
        die("يرجى تعبئة جميع الحقول المطلوبة.");
    }


    /* التأكد من عدم تكرار الرقم الوظيفي */
    $check = $conn->prepare("
        SELECT request_number
        FROM requests
        WHERE employee_number = ?
        AND request_number != ?
        LIMIT 1
    ");

    $check->bind_param(
        "si",
        $employee_number,
        $request_number
    );

    $check->execute();

    $checkResult = $check->get_result();

    if ($checkResult->num_rows > 0) {

        $check->close();

        die("الرقم الوظيفي مستخدم في طلب آخر.");

    }

    $check->close();


    /* تحديث الطلب */

    $stmt = $conn->prepare("
        UPDATE requests
        SET
            full_name = ?,
            employee_number = ?,
            category = ?,
            rank = ?,
            unit = ?,
            location = ?,
            notes = ?
        WHERE request_number = ?
    ");

    if (!$stmt) {
        die("حدث خطأ في تجهيز التعديل: " . $conn->error);
    }

    $stmt->bind_param(
        "sssssssi",
        $full_name,
        $employee_number,
        $category,
        $rank,
        $unit,
        $location,
        $notes,
        $request_number
    );

    if ($stmt->execute()) {

        $stmt->close();

        echo "
        <script>
            alert('تم تعديل بيانات الطلب بنجاح.');
            window.location.href = 'view-request.php?id=$request_number';
        </script>
        ";

        exit;

    } else {

        die("حدث خطأ أثناء تعديل الطلب: " . $stmt->error);

    }
}


/* =========================
   جلب بيانات الطلب
========================= */

$stmt = $conn->prepare("
    SELECT
        request_number,
        full_name,
        employee_number,
        category,
        rank,
        unit,
        location,
        notes
    FROM requests
    WHERE request_number = ?
    LIMIT 1
");

$stmt->bind_param("i", $request_number);
$stmt->execute();

$result = $stmt->get_result();

if ($result->num_rows === 0) {
    die("الطلب غير موجود.");
}

$request = $result->fetch_assoc();

$stmt->close();
