/* =====================================================

===================================================== */

function getStatusCount($conn, $status)
{
    $stmt = $conn->prepare(
        "SELECT COUNT(*) AS total
         FROM requests
         WHERE status = ?"
    );

    $stmt->bind_param("s", $status);
    $stmt->execute();

    $result = $stmt->get_result();
    $row = $result->fetch_assoc();

    $stmt->close();

    return (int)$row["total"];
}

   توزيع المواقع
===================================================== */

$locations = [];

$result = $conn->query("
    SELECT
        location,
        COUNT(*) AS total
    FROM requests
    GROUP BY location
    ORDER BY total DESC
");

    while ($row = $result->fetch_assoc()) {

        $locations[] = $row;

}


/* =====================================================
   توزيع الفئات
===================================================== */

$categories = [];

    SELECT
        category,
        COUNT(*) AS total
    FROM requests
    GROUP BY category

");

    SELECT COUNT(*) AS total

    WHERE request_date IS NOT NULL

      AND DATE_ADD(request_date, INTERVAL 6 MONTH)
