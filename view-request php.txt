require_once "db.php";

/* الحصول على رقم الطلب من الرابط */
$request_number = $_GET["id"] ?? "";

if ($request_number === "") {
    die("رقم الطلب غير موجود.");
}

/* جلب بيانات الطلب */
$stmt = $conn->prepare("
    SELECT
        request_number,
        full_name,
        employee_number,
        category,
        rank,
        unit,
        location,
        status,
        request_date,
        notification_date,
        housing_date,
        evacuation_date,
        notes
    FROM requests
    WHERE request_number = ?
    LIMIT 1
");

$stmt->bind_param("i", $request_number);
$stmt->execute();

$result = $stmt->get_result();

if ($result->num_rows === 0) {
    die("الطلب غير موجود.");
}

$request = $result->fetch_assoc();

$stmt->close();


/* تحويل حالة الطلب إلى كلاس CSS */
$statusClass = "status-new";

switch ($request["status"]) {

    case "تم الإشعار":
        $statusClass = "status-notified";
        break;

    case "تم التسكين":
        $statusClass = "status-housed";
        break;

    case "تم الإخلاء":
        $statusClass = "status-evacuated";
        break;

    case "لم يراجع":
        $statusClass = "status-not-reviewed";
        break;

    case "رفض السكن":
        $statusClass = "status-refused";
        break;
}


/* تنسيق التاريخ */
function formatDate($date)
{
    if (!$date) {
        return "لم يتم التسجيل";
    }

    return date("Y/m/d", strtotime($date));
}
