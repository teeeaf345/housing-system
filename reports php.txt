/* =========================
   إحصائيات الطلبات
========================= */

$total = 0;
$new = 0;
$notified = 0;
$housed = 0;
$evacuated = 0;
$notReviewed = 0;
$refused = 0;


/* العدد الإجمالي */

$result = $conn->query("
    SELECT COUNT(*) AS total
    FROM requests
");

if ($result) {

    $row = $result->fetch_assoc();

    $total = (int)$row["total"];
}


/* عدد الطلبات حسب الحالة */

$result = $conn->query("
    SELECT status, COUNT(*) AS total
    FROM requests
    GROUP BY status
");

if ($result) {

    while ($row = $result->fetch_assoc()) {

        switch ($row["status"]) {

            case "جديد":
                $new = (int)$row["total"];
                break;

            case "تم الإشعار":
                $notified = (int)$row["total"];
                break;

            case "تم التسكين":
                $housed = (int)$row["total"];
                break;

            case "تم الإخلاء":
                $evacuated = (int)$row["total"];
                break;

            case "لم يراجع":
                $notReviewed = (int)$row["total"];
                break;

            case "رفض السكن":
                $refused = (int)$row["total"];
                break;
        }
    }
}


/* =========================
   توزيع المواقع
========================= */

$locations = [];

$result = $conn->query("
    SELECT location, COUNT(*) AS total
    FROM requests
    GROUP BY location
    ORDER BY total DESC
");

if ($result) {

    while ($row = $result->fetch_assoc()) {

        $locations[] = $row;
    }
}


/* =========================
   توزيع الفئات
========================= */

$categories = [];

$result = $conn->query("
    SELECT category, COUNT(*) AS total
    FROM requests
    GROUP BY category
    ORDER BY total DESC
");

if ($result) {

    while ($row = $result->fetch_assoc()) {

        $categories[] = $row;
    }
}

    echo "رفض السكن\t" . $refused . "\n";

    echo "\nتوزيع المواقع\n";
    echo "الموقع\tالعدد\n";

        echo $location["location"] . "\t";
        echo $location["total"] . "\n";
    }

    echo "\nتوزيع الفئات\n";
    echo "الفئة\tالعدد\n";

    foreach ($categories as $category) {

        echo $category["category"] . "\t";
        echo $category["total"] . "\n";
    }

    exit;
}
