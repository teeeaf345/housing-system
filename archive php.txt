    جلب الطلبات المؤرشفة من قاعدة البيانات

    الأرشيف هنا يشمل:
    - تم الإخلاء
    - رفض السكن
*/

$sql = "
    SELECT
        request_number,
        full_name,
        employee_number,
        category,
        rank,
        unit,
        location,
        status,
        request_date,
        evacuation_date
    FROM requests
    WHERE status IN ('تم الإخلاء', 'رفض السكن')
    ORDER BY request_date DESC
";

$result = $conn->query($sql);

if (!$result) {
    die("حدث خطأ أثناء جلب بيانات الأرشيف: " . $conn->error);
}


/* تنسيق التاريخ */

function formatDate($date)
{
    if (!$date) {
        return "غير مسجل";
    }

    return date("Y/m/d", strtotime($date));
}


/* تحديد كلاس الحالة */

function getStatusClass($status)
{
    switch ($status) {

        case "تم الإخلاء":
            return "status-evacuated";

        case "رفض السكن":
            return "status-refused";

        default:
            return "status-new";
    }
}
