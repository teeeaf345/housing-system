/* تحديد كلاس الحالة */
function getStatusClass($status)
{
    switch ($status) {

        case "تم الإشعار":
            return "status-notified";

        case "تم التسكين":
            return "status-housed";

        case "تم الإخلاء":
            return "status-evacuated";

        case "لم يراجع":
            return "status-not-reviewed";

        case "رفض السكن":
            return "status-refused";

        default:
            return "status-new";
    }
}


/* تنسيق التاريخ */
function formatDate($date)
{
    if (!$date) {
        return "";
    }

    return date("Y/m/d", strtotime($date));
}

<script>

const searchName =
    document.getElementById("searchName");

const searchNumber =
    document.getElementById("searchNumber");

const statusFilter =
    document.getElementById("statusFilter");

const locationFilter =
    document.getElementById("locationFilter");

const rankFilter =
    document.getElementById("rankFilter");

const dateSort =
    document.getElementById("dateSort");

const tableBody =
    document.getElementById("requestsTableBody");

const resultsCount =
    document.getElementById("resultsCount");


function filterRequests() {

    const nameValue =
        searchName.value.trim().toLowerCase();

    const numberValue =
        searchNumber.value.trim();

    const statusValue =
        statusFilter.value;

    const locationValue =
        locationFilter.value;

    const rankValue =
        rankFilter.value;


    const rows =
        Array.from(
            tableBody.querySelectorAll("tr")
        );


    let visibleRows = 0;


    rows.forEach(row => {

        if (!row.dataset.name) {
            return;
        }


        const name =
            row.dataset.name.toLowerCase();

        const number =
            row.dataset.number;

        const status =
            row.dataset.status;

        const location =
            row.dataset.location;

        const rank =
            row.dataset.rank;


        const matchName =
            name.includes(nameValue);

        const matchNumber =
            number.includes(numberValue);

        const matchStatus =
            !statusValue ||
            status === statusValue;

        const matchLocation =
            !locationValue ||
            location === locationValue;

        const matchRank =
            !rankValue ||
            rank === rankValue;


        const show =
            matchName &&
            matchNumber &&
            matchStatus &&
            matchLocation &&
            matchRank;


        row.style.display =
            show ? "" : "none";


        if (show) {
            visibleRows++;
        }

    });


    resultsCount.textContent =
        visibleRows + " طلبات";

}


function sortRequests() {

    const rows =
        Array.from(
            tableBody.querySelectorAll("tr[data-date]")
        );


    rows.sort((a, b) => {

        const dateA =
            new Date(a.dataset.date);

        const dateB =
            new Date(b.dataset.date);


        if (dateSort.value === "newest") {

            return dateB - dateA;

        }

        return dateA - dateB;

    });


    rows.forEach(row => {

        tableBody.appendChild(row);

    });


    filterRequests();

}


function resetFilters() {

    searchName.value = "";

    searchNumber.value = "";

    statusFilter.value = "";

    locationFilter.value = "";

    rankFilter.value = "";

    dateSort.value = "oldest";


    sortRequests();

}


searchName.addEventListener(
    "input",
    filterRequests
);

searchNumber.addEventListener(
    "input",
    filterRequests
);

statusFilter.addEventListener(
    "change",
    filterRequests
);

locationFilter.addEventListener(
    "change",
    filterRequests
);

rankFilter.addEventListener(
    "change",
    filterRequests
);

dateSort.addEventListener(
    "change",
    sortRequests
);


sortRequests();

</script>
